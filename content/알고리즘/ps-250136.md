import java.io.*;
import java.util.*;

class Solution {
    static int[][] land;
    static int idx = 2;
    static ArrayList<Integer> l = new ArrayList<>(); // -2 해서 get하기
    static int[] dx={-1, 1, 0, 0}, dy={0, 0, -1, 1};
    
    public int solution(int[][] land) {
        Solution.land = land;
        for(int i=0; i<land.length; i++){
            for(int j=0; j<land[0].length; j++){
                if(land[i][j] == 1) l.add(bfs(i, j));
            }
        }
        
        int max = 0;
        for(int j=0; j<land[0].length; j++){
            int cur = 0;
            boolean[] visited = new boolean[idx];
            for(int i=0; i<land.length; i++){
                if(land[i][j] > 1 && !visited[land[i][j]]){
                    visited[land[i][j]] = true;
                    cur += l.get(land[i][j] - 2);
                }
            }
            max = Math.max(cur, max);
        }
        
        return max;
    }
    
    static public int bfs(int r, int c){
        Queue<int[]> q = new LinkedList<>();
        q.offer(new int[]{r, c});
        land[r][c] = idx;
        int size = 1;
        
        while(!q.isEmpty()){
            int[] cur = q.poll();
            int x = cur[0];
            int y = cur[1];
            
            for(int d=0; d<4; d++){
                int nx = x+dx[d];
                int ny = y+dy[d];
                if(nx>=0 && ny>=0 && nx<land.length && ny<land[0].length){
                    if(land[nx][ny] == 1){
                        land[nx][ny] = idx;
                        size++;
                        q.offer(new int[]{nx, ny});
                    }
                }
            }
        }
        
        idx++;
        return size;
    }
}
