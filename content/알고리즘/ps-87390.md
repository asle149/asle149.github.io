풀이:
배열을 만드는 것이 아니라 특정 부분만 계산하기!
```java
import java.io.*;
import java.util.*;

class Solution {
    public int[] solution(int n, long left, long right) {
        int[] ans = new int[(int)(right-left+1)];
        int idx = 0;
        
        for(long i=left; i<=right; i++){
            int num = Math.max((int)(i/n), (int)(i%n))+1;
            ans[idx] = num;
            idx++;
        }
        
        return ans;
    }
}
```
