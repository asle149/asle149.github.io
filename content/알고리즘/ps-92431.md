import java.io.*;
import java.util.*;

class Solution {
    public int[] solution(int[] fees, String[] records) {
        int bt = fees[0]; int bf = fees[1];
        int ut = fees[2]; int uf = fees[3];
        ArrayList<Integer> l = new ArrayList<>();
        
        String[][] car = new String[records.length][3];
        for(int i=0; i<records.length; i++){
            car[i] = records[i].split(" ");
        }
        
        Arrays.sort(car, (o1, o2)->{
            return o1[1].compareTo(o2[1]);
        });
        
        for(int i=0; i<car.length; i++){
            int start = cvt(car[i][0]);
            int time = 0;
            
            while(i+1<car.length && car[i][1].equals(car[i+1][1])){
                if(car[i+1][2].equals("OUT")){
                    time += cvt(car[i+1][0])-start;
                    start = -1;
                }else{
                    start = cvt(car[i+1][0]);
                }
                i++;
            }
            
            if(start != -1) time += cvt("23:59")-start;
            l.add(time);
        }
        
        int[] ans = new int[l.size()];
        for(int i=0; i<l.size(); i++){
            if(l.get(i)-bt<=0) ans[i] = bf;
            else {
                ans[i] = (int) (bf + (Math.ceil((double)((l.get(i)-bt))/ut))*uf);
            }
        }

        return ans;
    }
    
    static public int cvt(String str){
        String[] a = str.split(":");
        return Integer.parseInt(a[0])*60+Integer.parseInt(a[1]);
    }
}
