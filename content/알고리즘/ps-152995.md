```java
import java.io.*;
import java.util.*;

class Solution {
    public int solution(int[][] arr) {
        int scoreA = arr[0][0];
        int scoreB = arr[0][1];
        Arrays.sort(arr, (a1, a2) -> {
            if (a1[0] == a2[0]) {
                return a1[1] - a2[1]; 
            }
            return a2[0] - a1[0];
        });
        
                                      
        int max = arr[0][1];
        List<int[]> l = new ArrayList<>();
        l.add(new int[]{arr[0][0], arr[0][1], arr[0][0] + arr[0][1]});
        for(int i=1; i<arr.length; i++){
            if(max>arr[i][1] && arr[i][0] != arr[0][0]) continue;
            else{
                if(max<arr[i][1]) max = arr[i][1];
                l.add(new int[]{arr[i][0], arr[i][1], arr[i][0]+arr[i][1]});
            }
        }
        
        l.sort((a1, a2) -> a2[2] - a1[2]);
        if(l.get(0)[0] == scoreA && l.get(0)[1] == scoreB) return 1;
        int cnt = 1;
        int before = l.get(0)[2];
        for(int i=1; i<l.size(); i++){
            int[] cur = l.get(i);
            if(before != cur[2]) cnt = i+1;
            before = cur[2];
            if(cur[0] == scoreA && cur[1] == scoreB) return cnt;
        }
        
        return -1;
    }
}
```

간소화

- **목록 `l`과 두 번째 정렬 없애기**  
    인센티브 대상자 중 **점수 합이 완호보다 큰 사람 수**만 세면 됨.
    
- **제외 조건 단순화하기**  
    지금 정렬 순서에서는 `max > arr[i][1]`만 확인해도 됨

```java
import java.util.Arrays;

class Solution {
    public int solution(int[][] scores) {
        int wanhoA = scores[0][0];
        int wanhoB = scores[0][1];
        int wanhoSum = wanhoA + wanhoB;

        // 첫 점수 내림차순, 같으면 두 번째 점수 오름차순
        Arrays.sort(scores, (a, b) -> {
            if (a[0] == b[0]) {
                return Integer.compare(a[1], b[1]);
            }
            return Integer.compare(b[0], a[0]);
        });

        int maxB = 0;
        int rank = 1;

        for (int[] score : scores) {
            // 앞사람 중 두 점수가 모두 높은 사람이 있음
            if (score[1] < maxB) {
                if (score[0] == wanhoA && score[1] == wanhoB) {
                    return -1;
                }
                continue;
            }

            maxB = score[1];

            // 인센티브 대상 중 완호보다 합계가 높은 사람
            if (score[0] + score[1] > wanhoSum) {
                rank++;
            }
        }

        return rank;
    }
}
```
