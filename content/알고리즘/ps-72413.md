1) 다익스트라
```java
import java.io.*;
import java.util.*;


class Solution {
    static class Edge{
        int to, dis;
        Edge(int to, int dis){
            this.to = to; this.dis = dis;
        }
    }
    
    static int n;
    static List<Edge>[] l;
    
    public int solution(int n, int s, int a, int b, int[][] fares) {
        Solution.n = n;
        l = new ArrayList[n+1];
        for(int i=0; i<=n; i++) l[i] = new ArrayList<>();
        
        for(int i=0; i<fares.length; i++){
            int n1 = fares[i][0];
            int n2 = fares[i][1];
            int c = fares[i][2];
            l[n1].add(new Edge(n2, c));
            l[n2].add(new Edge(n1, c));
        }
        
        int[] distS = dijkstra(s);
        int[] distA = dijkstra(a);
        int[] distB = dijkstra(b);
        
        int ans = Integer.MAX_VALUE;
        for(int i=0; i<=n; i++){
            if(distS[i] == Integer.MAX_VALUE || distA[i] == Integer.MAX_VALUE
                || distB[i] == Integer.MAX_VALUE) continue;
            ans = Math.min(ans, distS[i] + distA[i] + distB[i]);
        }
        
        return ans;
    }
  
    public static int[] dijkstra(int start){
        PriorityQueue<int[]> pq = new PriorityQueue<>((a, b)->(a[1]-b[1]));
        int[] dist = new int[n+1];
        Arrays.fill(dist, Integer.MAX_VALUE);
        dist[start] = 0;
        pq.offer(new int[]{start, 0});
        
        while(!pq.isEmpty()){
            int[] cur = pq.poll();
            int node = cur[0];
            int d = cur[1];
            if (d > dist[node]) continue;
            for(Edge e : l[node]){
                if(dist[e.to] <= d + e.dis) continue;
                dist[e.to] = d + e.dis;
                pq.offer(new int[]{e.to, dist[e.to]});
            }
        }
        
        return dist;
    }
}
```

2) 플로이드 워셜
```java
import java.util.Arrays;

class Solution {
    public int solution(int n, int s, int a, int b, int[][] fares) {
        // 아직 도달할 수 없다는 표시입니다.
        // 이 문제에서 가능한 실제 최소 요금보다 충분히 큰 값입니다.
        final int INF = 1_000_000_000;

        // dist[i][j]: i에서 j까지 현재까지 구한 최소 요금
        int[][] dist = new int[n + 1][n + 1];

        for (int i = 1; i <= n; i++) {
            // Arrays 클래스의 fill 메서드로 각 행을 채웁니다.
            Arrays.fill(dist[i], INF);

            // 자기 자신까지 이동하는 비용은 0입니다.
            dist[i][i] = 0;
        }

        // 직접 연결된 도로의 요금을 저장합니다.
        for (int[] fare : fares) {
            int from = fare[0];
            int to = fare[1];
            int cost = fare[2];

            // 양방향 도로이므로 양쪽에 저장합니다.
            dist[from][to] = cost;
            dist[to][from] = cost;
        }

        // k: 이번에 경유할 수 있도록 추가하는 지점
        // k까지 처리하면 1~k를 중간에 거치는 최단 경로가 반영됩니다.
        for (int k = 1; k <= n; k++) {

            // i: 출발 지점
            for (int i = 1; i <= n; i++) {
                // i에서 k로 갈 수 없으면 k를 경유할 수 없습니다.
                if (dist[i][k] == INF) continue;

                // j: 도착 지점
                for (int j = 1; j <= n; j++) {
                    // k에서 j로 갈 수 없어도 경유할 수 없습니다.
                    if (dist[k][j] == INF) continue;

                    // 기존 경로와 i → k → j 경로 중 싼 것을 저장합니다.
                    dist[i][j] = Math.min(
                        dist[i][j],
                        dist[i][k] + dist[k][j]
                    );
                }
            }
        }

        int answer = INF;

        // 여기서 k는 두 사람이 갈라지는 지점입니다.
        for (int k = 1; k <= n; k++) {
            // 필요한 세 구간 중 하나라도 이동 불가능하면 제외합니다.
            if (dist[s][k] == INF
                    || dist[k][a] == INF
                    || dist[k][b] == INF) {
                continue;
            }

            // s → k: 함께 이동하므로 요금을 한 번만 더합니다.
            // k → a, k → b: 각자 이동하는 요금을 더합니다.
            int total = dist[s][k] + dist[k][a] + dist[k][b];

            answer = Math.min(answer, total);
        }

        // k = s인 경우도 비교하므로 합승하지 않는 경우까지 포함됩니다.
        return answer;
    }
}
```
