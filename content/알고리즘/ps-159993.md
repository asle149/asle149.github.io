import java.io.*;
import java.util.*;

class Solution {
    static int[] dx={-1, 1, 0, 0}, dy={0, 0, -1, 1};
    
    public int solution(String[] m) {
        char[][] maps = new char[m.length][m[0].length()];
        for(int i=0; i<m.length; i++){
            maps[i] = m[i].toCharArray();
        }
        
        int sx=0, sy=0, ex=0, ey=0;
        for(int i=0; i<maps.length; i++){
            for(int j=0; j<maps[0].length; j++){
                if(maps[i][j] == 'S'){sx = i; sy = j;}
                else if(maps[i][j] == 'E'){ex = i; ey = j;}
            }
        }
        
        int ans = bfs(sx, sy, ex, ey, maps);
        
        return ans;
    }
    // 0:x 1:y 2:레버잇없 3:최단거리
    
    public static int bfs(int sx, int sy, int ex, int ey, char[][] maps){
        Queue<int[]> q = new LinkedList<>();
        boolean[][][] visited = new boolean[maps.length][maps[0].length][2];
        q.offer(new int[]{sx, sy, 0, 0});
        visited[sx][sy][0] = true;
        
        while(!q.isEmpty()){
            int[] cur = q.poll();
            int x = cur[0]; int y = cur[1]; int l = cur[2]; int d = cur[3];
            if(x==ex && y==ey && l==1) return d;
            
            for(int t=0; t<4; t++){
                int nx = x+dx[t];
                int ny = y+dy[t];
                if(nx>=0 && nx<maps.length && ny>=0 && ny<maps[0].length && maps[nx][ny]!='X'){
                    int nextLever = l;
                    if (maps[nx][ny] == 'L') nextLever = 1;
                    if(!visited[nx][ny][nextLever]){
                        visited[nx][ny][nextLever] = true;
                        q.offer(new int[]{nx, ny, nextLever, d+1});
                    }
                }
            }
        }
        
        return -1;
    }
}
