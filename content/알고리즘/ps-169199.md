import java.io.*;
import java.util.*;


class Solution {
    static int rx, ry, gx, gy;
    static int[] dx = {-1, 1, 0, 0}, dy = {0, 0, -1, 1};
    
    static class Node{
        int x, y, cnt; //도착하는 x, y, 이동횟수
        Node(int x, int y, int cnt) {
            this.x = x; this.y = y; 
            this.cnt = cnt;
        }
    }
    
    public int solution(String[] board) {
        char[][] b = new char[board.length][board[0].length()];
        for(int i=0; i<board.length; i++) b[i] = board[i].toCharArray();
        for(int i=0; i<b.length; i++){
            for(int j=0; j<b[0].length; j++){
                if(b[i][j] == 'R'){rx = i; ry = j;}
                if(b[i][j] == 'G'){gx = i; gy = j;}
            }
        }
        
        int ans = bfs(b);
        
        return ans;
    }
    
    static int bfs(char[][] b){
        Queue<Node> q = new LinkedList<>();
        boolean visited[][] = new boolean[b.length][b[0].length];
        q.offer(new Node(rx, ry, 0));
        visited[rx][ry] = true;
        
        while(!q.isEmpty()){
            Node cur = q.poll();
            int x = cur.x; int y = cur.y; int c = cur.cnt;
            if(x == gx && y == gy) return c;
            for(int d=0; d<4; d++){
                int nx = x; int ny = y;
                while (true) {
                    int nextX = nx + dx[d];
                    int nextY = ny + dy[d];
                    if (nextX < 0 || nextX >= b.length
                        || nextY < 0 || nextY >= b[0].length
                        || b[nextX][nextY] == 'D') break;
                    nx = nextX;
                    ny = nextY;
                }
                
                if((nx!=x || ny!=y) && !visited[nx][ny]){
                    visited[nx][ny] = true;
                    q.offer(new Node(nx, ny, c+1));
                }
            }
        }
        
        return -1;
    }
}
