풀이:
`Integer.toString(num, base)`를 까먹지 말자!!

```java
import java.io.*;
import java.util.*;

class Solution {
    public String solution(int n, int t, int m, int p) {
        char[] arr = new char[t*m+1];
        arr[1] = '0';
        int num = 0;
        
        for(int i=2; i<arr.length;){
            num++;
            String str = Integer.toString(num, n).toUpperCase();
            for(int j=0; j<str.length() && i<arr.length; j++){
                arr[i] = str.charAt(j);
                i++;
            }
        }
        
        StringBuilder sb = new StringBuilder();
        for(int i=1; i<t*m+1; i+=m){
            sb.append(arr[i+p-1]);
        }
        
        return sb.toString();
    }
}
```
