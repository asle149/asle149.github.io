import java.io.*;
import java.util.*;

class Solution {
    public int solution(int n, int[][] wires) {
        boolean[][] arr = new boolean[n+1][n+1];
        
        for(int i=0; i<wires.length; i++){
            int a = wires[i][0];
            int b = wires[i][1];
            arr[a][b] = true;
            arr[b][a] = true;
        }
        
        int diff = Integer.MAX_VALUE;
        
        for(int i=0; i<wires.length; i++){
            int a = wires[i][0];
            int b = wires[i][1];
            arr[a][b] = false;
            arr[b][a] = false;
            
            int cnt = bfs(arr, n, a);
            diff = Math.min(Math.abs(2*cnt - n), diff);
            
            arr[a][b] = true;
            arr[b][a] = true;
        }
        
        return diff;
    }
    
    public static int bfs(boolean[][] arr, int n, int s){
        boolean[] visited = new boolean[n+1];
        Queue<Integer> q = new LinkedList<>();
        q.offer(s); visited[s] = true;
        int cnt = 1;
        
        while(!q.isEmpty()){
            int cur = q.poll();
            for(int i=1; i<=n; i++){
                if(arr[cur][i] && !visited[i]){
                    visited[i] = true;
                    q.offer(i);
                    cnt++;
                }
            }
        }
        
        return cnt;
    }
}
