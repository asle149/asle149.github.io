import java.io.*;
import java.util.*;

class Solution {
    public int[] solution(String msg) {
        ArrayList<String> l = new ArrayList<>();
        ArrayList<Integer> ans = new ArrayList<>();
        
        for(int i=65; i<65+26; i++){
            l.add(String.valueOf((char) i));
        }
        
        for(int i=0; i<msg.length();){
            int num = 1;
            StringBuilder sb = new StringBuilder();

            sb.append(msg.charAt(i));
            int idx = l.indexOf(sb.toString())+1;
            
            while(i + num < msg.length()){
                sb.append(msg.charAt(i+num));
                String str = sb.toString();
                if(!l.contains(str)){
                    l.add(str);
                    break;
                }
                idx = l.indexOf(sb.toString())+1;
                num++;
            }
            ans.add(idx);
            i += num;
        }
        
        Integer[] resI = ans.toArray(new Integer[ans.size()]);
        int[] res = Arrays.stream(resI).mapToInt(Integer::intValue).toArray();
        
        return res;
    }
}
