import java.io.*;
import java.util.*;

class Solution {
    static class Node{
        int x, y, cost, dir;
        Node(int x, int y, int cost, int dir){
            this.x = x; this.y = y;
            this.cost = cost; this.dir = dir; //0:상 1:하 2:좌 3:우
        } 
    }
    
    public int solution(int[][] board) {
        int n = board.length;
        
        PriorityQueue<Node> pq = 
            new PriorityQueue<>((a, b) -> Integer.compare(a.cost, b.cost));
        pq.offer(new Node(0, 0, 0, 0)); pq.offer(new Node(0, 0, 0, 1));
        pq.offer(new Node(0, 0, 0, 2)); pq.offer(new Node(0, 0, 0, 3));
        
        int[][][] minCost = new int[n][n][4];
        for(int i=0; i<n; i++){
            for(int j=0; j<n; j++){
                for(int k=0; k<4; k++) minCost[i][j][k] = Integer.MAX_VALUE;
            }
        }
        minCost[0][0][0] = 0; minCost[0][0][1] = 0;
        minCost[0][0][2] = 0; minCost[0][0][3] = 0;
        
        int[] dx = {-1, 1, 0, 0}, dy = {0, 0, -1, 1};
        
        while(!pq.isEmpty()){
            Node cur = pq.poll();
            int x = cur.x; int y = cur.y;
            int cost = cur.cost; int dir = cur.dir;
            
            for(int d=0; d<4; d++){
                int nx = x+dx[d];
                int ny = y+dy[d];
                
                if(nx>=0 && ny>=0 && nx<n && ny<n){
                    if(board[nx][ny] == 0){
                        int extra = 100;
                        if(d != dir) extra += 500;
                        if((cost+extra)<minCost[nx][ny][d]){
                            minCost[nx][ny][d] = cost+extra;
                            pq.offer(new Node(nx, ny, minCost[nx][ny][d], d));
                        }
                    }
                }
            }
        }
        
        int ans = minCost[n-1][n-1][0];
        for(int i=1; i<4; i++) ans = Math.min(ans, minCost[n-1][n-1][i]);
        
        return ans;
    }
}
